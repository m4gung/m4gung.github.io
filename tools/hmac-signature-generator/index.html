💻 Kode Verifikasi HMAC di Berbagai Bahasa Backend

Java 8+ / javax.crypto.Mac (Standard JRE) 
import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.util.HexFormat; // Java 17+ atau gunakan apache commons Hex.encodeHexString

public class HmacUtils {

    public static String calculateHmacSHA256(String data, String key) throws Exception {
        Mac mac = Mac.getInstance("HmacSHA256");
        SecretKeySpec secretKeySpec = new SecretKeySpec(key.getBytes(StandardCharsets.UTF_8), "HmacSHA256");
        mac.init(secretKeySpec);
        byte[] hmacBytes = mac.doFinal(data.getBytes(StandardCharsets.UTF_8));
        
        // Java 17+:
        return HexFormat.of().formatHex(hmacBytes);
    }

    // Timing-safe equals untuk mencegah timing attack saat verifikasi webhook
    public static boolean verifySignature(String expectedSig, String data, String key) {
        try {
            String calculated = calculateHmacSHA256(data, key);
            return MessageDigest.isEqual(
                expectedSig.getBytes(StandardCharsets.UTF_8),
                calculated.getBytes(StandardCharsets.UTF_8)
            );
        } catch (Exception e) {
            return false;
        }
    }
}
Go (crypto/hmac & crypto/subtle) 
package main

import (
    "crypto/hmac"
    "crypto/sha256"
    "crypto/subtle"
    "encoding/hex"
)

// CalculateHmacSHA256 menghitung signature dalam format hex
func CalculateHmacSHA256(message, secret string) string {
    h := hmac.New(sha256.New, []byte(secret))
    h.Write([]byte(message))
    return hex.EncodeToString(h.Sum(nil))
}

// VerifySignature memverifikasi signature dengan timing-safe subtle.ConstantTimeCompare
func VerifySignature(expectedSig, message, secret string) bool {
    calculated := CalculateHmacSHA256(message, secret)
    return subtle.ConstantTimeCompare([]byte(expectedSig), []byte(calculated)) == 1
}
Python 3 (hmac & hashlib) 
import hmac
import hashlib

def calculate_hmac_sha256(message: str, secret: str) -> str:
    key_bytes = secret.encode('utf-8')
    msg_bytes = message.encode('utf-8')
    return hmac.new(key_bytes, msg_bytes, hashlib.sha256).hexdigest()

def verify_signature(expected_sig: str, message: str, secret: str) -> bool:
    calculated = calculate_hmac_sha256(message, secret)
    # Gunakan hmac.compare_digest untuk mencegah timing attack
    return hmac.compare_digest(expected_sig, calculated)
Node.js (crypto.createHmac & timingSafeEqual) 
const crypto = require('crypto');

function calculateHmacSHA256(message, secret) {
    return crypto
        .createHmac('sha256', secret)
        .update(message, 'utf8')
        .digest('hex');
}

function verifySignature(expectedSig, message, secret) {
    const calculated = calculateHmacSHA256(message, secret);
    
    // Cegah timing attacks dengan crypto.timingSafeEqual
    const expectedBuf = Buffer.from(expectedSig);
    const calculatedBuf = Buffer.from(calculated);
    
    if (expectedBuf.length !== calculatedBuf.length) {
        return false;
    }
    return crypto.timingSafeEqual(expectedBuf, calculatedBuf);
}
PHP (hash_hmac & hash_equals) 
<?php

function calculateHmacSHA256(string $message, string $secret): string {
    return hash_hmac('sha256', $message, $secret);
}

function verifySignature(string $expectedSig, string $message, string $secret): bool {
    $calculated = calculateHmacSHA256($message, $secret);
    // Selalu gunakan hash_equals untuk membandingkan hash/signature
    return hash_equals($expectedSig, $calculated);
}
