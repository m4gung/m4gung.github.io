📖 Rangkuman Spesifikasi SC-73 Bank Danamon v1.2

SNAP BI B2B Access Token

📌 General Specification

  • Endpoint URL: https://<ip>/api/v2.0/access-token/b2b (SIT, UAT, Production)
  • HTTP Method: POST
  • Content-Type: application/json
  • Security: Digital Signature Asymmetric (SHA256withRSA / RSASSA-PKCS1-v1_5)
  • Panjang Kunci: RSA 4096-bit (OpenSSL: openssl genpkey -algorithm RSA -out private.pem -pkeyopt rsa_keygen_bits:4096)

import java.nio.charset.StandardCharsets;
import java.security.KeyFactory;
import java.security.PrivateKey;
import java.security.PublicKey;
import java.security.Signature;
import java.security.spec.PKCS8EncodedKeySpec;
import java.security.spec.X509EncodedKeySpec;
import java.util.Base64;
import java.util.HexFormat; // Java 17+

public class SnapBiSignatureUtil {

    public static String generateSignatureHex(String clientId, String timestamp, String privateKeyPem) throws Exception {
        String stringToSign = clientId + "|" + timestamp;
        
        // Bersihkan header PEM
        String cleanKey = privateKeyPem
            .replace("-----BEGIN PRIVATE KEY-----", "")
            .replace("-----END PRIVATE KEY-----", "")
            .replaceAll("\\s+", "");
        
        byte[] keyBytes = Base64.getDecoder().decode(cleanKey);
        PKCS8EncodedKeySpec spec = new PKCS8EncodedKeySpec(keyBytes);
        KeyFactory kf = KeyFactory.getInstance("RSA");
        PrivateKey privateKey = kf.generatePrivate(spec);

        Signature signer = Signature.getInstance("SHA256withRSA");
        signer.initSign(privateKey);
        signer.update(stringToSign.getBytes(StandardCharsets.UTF_8));
        byte[] signatureBytes = signer.sign();

        // Format ke Hex (Java 17+)
        return HexFormat.of().formatHex(signatureBytes);
    }

    public static boolean verifySignatureHex(String clientId, String timestamp, String signatureHex, String publicKeyPem) throws Exception {
        String stringToSign = clientId + "|" + timestamp;
        
        String cleanKey = publicKeyPem
            .replace("-----BEGIN PUBLIC KEY-----", "")
            .replace("-----END PUBLIC KEY-----", "")
            .replaceAll("\\s+", "");
            
        byte[] keyBytes = Base64.getDecoder().decode(cleanKey);
        X509EncodedKeySpec spec = new X509EncodedKeySpec(keyBytes);
        KeyFactory kf = KeyFactory.getInstance("RSA");
        PublicKey publicKey = kf.generatePublic(spec);

        Signature verifier = Signature.getInstance("SHA256withRSA");
        verifier.initVerify(publicKey);
        verifier.update(stringToSign.getBytes(StandardCharsets.UTF_8));
        
        byte[] sigBytes = HexFormat.of().parseHex(signatureHex);
        return verifier.verify(sigBytes);
    }
}

const crypto = require('crypto');

function generateSignature(clientId, timestamp, privateKeyPem) {
    const stringToSign = `${clientId}|${timestamp}`;
    const signer = crypto.createSign('RSA-SHA256');
    signer.update(stringToSign);
    return signer.sign(privateKeyPem, 'hex'); // Output hex sesuai spek Danamon
}

function verifySignature(clientId, timestamp, signatureHex, publicKeyPem) {
    const stringToSign = `${clientId}|${timestamp}`;
    const verifier = crypto.createVerify('RSA-SHA256');
    verifier.update(stringToSign);
    return verifier.verify(publicKeyPem, Buffer.from(signatureHex, 'hex'));
}
Python (cryptography package): 
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric import padding
from cryptography.hazmat.primitives.serialization import load_pem_private_key, load_pem_public_key

def generate_signature(client_id: str, timestamp: str, private_key_pem: str) -> str:
    string_to_sign = f"{client_id}|{timestamp}".encode('utf-8')
    private_key = load_pem_private_key(private_key_pem.encode('utf-8'), password=None)
    
    signature = private_key.sign(
        string_to_sign,
        padding.PKCS1v15(),
        hashes.SHA256()
    )
    return signature.hex()

def verify_signature(client_id: str, timestamp: str, signature_hex: str, public_key_pem: str) -> bool:
    string_to_sign = f"{client_id}|{timestamp}".encode('utf-8')
    public_key = load_pem_public_key(public_key_pem.encode('utf-8'))
    try:
        public_key.verify(
            bytes.fromhex(signature_hex),
            string_to_sign,
            padding.PKCS1v15(),
            hashes.SHA256()
        )
        return True
    except Exception:
        return False
Go (crypto/rsa & crypto/sha256): 
package main

import (
	"crypto"
	"crypto/rand"
	"crypto/rsa"
	"crypto/sha256"
	"crypto/x509"
	"encoding/hex"
	"encoding/pem"
	"fmt"
)

func GenerateSignature(clientID, timestamp, privKeyPEM string) (string, error) {
	stringToSign := fmt.Sprintf("%s|%s", clientID, timestamp)
	block, _ := pem.Decode([]byte(privKeyPEM))
	privKey, err := x509.ParsePKCS8PrivateKey(block.Bytes)
	if err != nil {
		return "", err
	}

	rsaPriv, ok := privKey.(*rsa.PrivateKey)
	if !ok {
		return "", fmt.Errorf("not an RSA private key")
	}

	hash := sha256.Sum256([]byte(stringToSign))
	sigBytes, err := rsa.SignPKCS1v15(rand.Reader, rsaPriv, crypto.SHA256, hash[:])
	if err != nil {
		return "", err
	}

	return hex.EncodeToString(sigBytes), nil
}
PHP (OpenSSL extension): 
function generateSignature($clientId, $timestamp, $privateKeyPem) {
    $stringToSign = "{$clientId}|{$timestamp}";
    $privateKey = openssl_pkey_get_private($privateKeyPem);
    
    $binarySignature = "";
    openssl_sign($stringToSign, $binarySignature, $privateKey, OPENSSL_ALGO_SHA256);
    
    return bin2hex($binarySignature);
}

function verifySignature($clientId, $timestamp, $signatureHex, $publicKeyPem) {
    $stringToSign = "{$clientId}|{$timestamp}";
    $publicKey = openssl_pkey_get_public($publicKeyPem);
    $binarySignature = hex2bin($signatureHex);
    
    $ok = openssl_verify($stringToSign, $binarySignature, $publicKey, OPENSSL_ALGO_SHA256);
    return $ok === 1;
}
